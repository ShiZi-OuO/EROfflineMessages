cmake_minimum_required(VERSION 3.24)
project(EROfflineMessages VERSION 0.1.5 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(EROfflineObserver SHARED src/Observer.cpp)
target_compile_definitions(EROfflineObserver PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(EROfflineObserver PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(EROfflineObserver PRIVATE version bcrypt)
set_target_properties(EROfflineObserver PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(er_data STATIC src/MessagePack.cpp)
target_include_directories(er_data PUBLIC src PRIVATE third_party/json)
target_compile_options(er_data PRIVATE /W4 /utf-8 /EHsc)
add_executable(er_pack_check tools/pack_check.cpp)
target_link_libraries(er_pack_check PRIVATE er_data)
target_compile_options(er_pack_check PRIVATE /W4 /utf-8 /EHsc)
add_library(er_spatial STATIC src/SpatialIndex.cpp)
target_include_directories(er_spatial PUBLIC src)
target_compile_options(er_spatial PRIVATE /W4 /utf-8 /EHsc)
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
add_library(EROfflineMessages SHARED src/NativeReplay.cpp src/RuntimeSupport.cpp)
target_link_libraries(EROfflineMessages PRIVATE er_data er_spatial minhook bcrypt)
target_compile_definitions(EROfflineMessages PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(EROfflineMessages PRIVATE /W4 /utf-8 /EHsc)
set_target_properties(EROfflineMessages PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(EROfflineQueryProbe SHARED src/NativeQueryProbe.cpp src/RuntimeSupport.cpp)
target_link_libraries(EROfflineQueryProbe PRIVATE minhook bcrypt)
target_compile_definitions(EROfflineQueryProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(EROfflineQueryProbe PRIVATE /W4 /utf-8 /EHsc)
set_target_properties(EROfflineQueryProbe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(ERNativeRegionCapture SHARED src/NativeRegionCapture.cpp src/RuntimeSupport.cpp)
target_include_directories(ERNativeRegionCapture PRIVATE third_party/json)
target_link_libraries(ERNativeRegionCapture PRIVATE minhook bcrypt)
target_compile_definitions(ERNativeRegionCapture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(ERNativeRegionCapture PRIVATE /W4 /utf-8 /EHsc)
set_target_properties(ERNativeRegionCapture PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
if(BUILD_TESTING)
  add_executable(cache_budget_tests tests/cache_budget.cpp)
  target_include_directories(cache_budget_tests PRIVATE src)
  add_test(NAME native_cache_budget_guards COMMAND cache_budget_tests)
  add_executable(spatial_index_tests tests/spatial_index.cpp)
  target_link_libraries(spatial_index_tests PRIVATE er_spatial er_data)
  target_compile_options(spatial_index_tests PRIVATE /W4 /utf-8 /EHsc)
  add_test(NAME spatial_selection_guards COMMAND spatial_index_tests)
  add_executable(message_pack_tests tests/message_pack.cpp)
  target_include_directories(message_pack_tests PRIVATE third_party/json)
  target_link_libraries(message_pack_tests PRIVATE er_data)
  add_test(NAME message_pack_guards COMMAND message_pack_tests)
  add_executable(runtime_read_tests tests/runtime_reads.cpp src/RuntimeSupport.cpp)
  target_include_directories(runtime_read_tests PRIVATE src)
  target_compile_definitions(runtime_read_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(runtime_read_tests PRIVATE /W4 /utf-8 /EHsc)
  target_link_libraries(runtime_read_tests PRIVATE bcrypt)
  add_test(NAME runtime_address_guards COMMAND runtime_read_tests)
  add_executable(replay_refusal_tests tests/replay_refusal.cpp)
  target_compile_definitions(replay_refusal_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(replay_refusal_tests PRIVATE /W4 /utf-8 /EHsc)
  add_dependencies(replay_refusal_tests EROfflineMessages)
  add_test(NAME replay_unsupported_and_duplicate_host COMMAND replay_refusal_tests "$<TARGET_FILE:EROfflineMessages>" "${CMAKE_BINARY_DIR}/test-runs")
endif()
